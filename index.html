using System.Collections; using System.Collections.Generic; using UnityEngine; using UnityEngine.AI; public class GameControllerMobile : MonoBehaviour { [Header("--- MOVIMENTAÇÃO MOBILE ---")] public CharacterController controller; public Transform cameraTransform; public float moveSpeed = 4.0f; public float lookSensitivity = 0.2f; public float gravity = -9.81f; private Vector3 velocity; private float xRotation = 0f; private int leftTouchId = -1; private int rightTouchId = -1; [Header("--- VIDA DO JOGADOR ---")] public float maxHealth = 100f; public float currentHealth; public bool isPlayerDead = false; [Header("--- SISTEMA DE ARMAS ---")] public float gunDamage = 25f; public float gunRange = 100f; public float fireRate = 0.2f; public int maxAmmo = 30; public float reloadTime = 2.0f; public ParticleSystem muzzleFlash; private int currentAmmo; private float nextTimeToFire = 0f; private bool isReloading = false; [Header("--- INTERAÇÃO (PORTAS/JANELAS) ---")] public float interactDistance = 3.0f; void Start() { if (controller == null) controller = GetComponent(); currentHealth = maxHealth; currentAmmo = maxAmmo; } void Update() { if (isPlayerDead) return; HandleMobileControls(); ApplyGravity(); } // ========================================== // CONTROLES E CÂMERA MOBILE // ========================================== private void HandleMobileControls() { for (int i = 0; i < Input.touchCount; i++) { Touch touch = Input.GetTouch(i); // Metade Esquerda da Tela: Movimentação if (touch.position.x < Screen.width / 2) { if (touch.phase == TouchPhase.Began) leftTouchId = touch.fingerId; if (touch.fingerId == leftTouchId) { Vector2 delta = touch.deltaPosition; Vector3 move = transform.right * delta.x + transform.forward * delta.y; controller.Move(move * moveSpeed * Time.deltaTime); if (touch.phase == TouchPhase.Ended || touch.phase == TouchPhase.Canceled) leftTouchId = -1; } } // Metade Direita da Tela: Controle da Câmera else { if (touch.phase == TouchPhase.Began) rightTouchId = touch.fingerId; if (touch.fingerId == rightTouchId && touch.phase == TouchPhase.Moved) { float mouseX = touch.deltaPosition.x * lookSensitivity; float mouseY = touch.deltaPosition.y * lookSensitivity; xRotation -= mouseY; xRotation = Mathf.Clamp(xRotation, -80f, 80f); if (cameraTransform != null) cameraTransform.localRotation = Quaternion.Euler(xRotation, 0f, 0f); transform.Rotate(Vector3.up * mouseX); if (touch.phase == TouchPhase.Ended || touch.phase == TouchPhase.Canceled) rightTouchId = -1; } } } } private void ApplyGravity() { if (controller.isGrounded && velocity.y < 0) { velocity.y = -2f; } velocity.y += gravity * Time.deltaTime; controller.Move(velocity * Time.deltaTime); } // ========================================== // SISTEMA DE TIRO E RECARGA // Vincule esta função a um botão de interface (UI Button) // ========================================== public void ShootButton() { if (isReloading || currentAmmo <= 0 || Time.time < nextTimeToFire || isPlayerDead) return; nextTimeToFire = Time.time + fireRate; currentAmmo--; if (muzzleFlash != null) muzzleFlash.Play(); RaycastHit hit; if (Physics.Raycast(cameraTransform.position, cameraTransform.forward, out hit, gunRange)) { // Tenta causar dano no Zumbi atingido ZombieEntity zombie = hit.transform.GetComponent(); if (zombie != null) { zombie.TakeDamage(gunDamage); } } if (currentAmmo <= 0) { StartCoroutine(ReloadWeapon()); } } private IEnumerator ReloadWeapon() { isReloading = true; yield return new WaitForSeconds(reloadTime); currentAmmo = maxAmmo; isReloading = false; } // ========================================== // INTERAÇÃO COM PORTAS E JANELAS // Vincule esta função a um botão de interagir na tela // ========================================== public void InteractButton() { RaycastHit hit; if (Physics.Raycast(cameraTransform.position, cameraTransform.forward, out hit, interactDistance)) { InteractiveObject doorOrWindow = hit.transform.GetComponent(); if (doorOrWindow != null) { doorOrWindow.ToggleState(); } } } // ========================================== // RECEBER DANO DOS ZUMBIS // ========================================== public void PlayerTakeDamage(float damageAmount) { if (isPlayerDead) return; currentHealth -= damageAmount; if (currentHealth <= 0) { currentHealth = 0; isPlayerDead = true; Debug.Log("GAME OVER: O jogador morreu!"); } } } // ========================================== // CLASSE DOS ZUMBIS (Anexe aos Inimigos 3D) // ========================================== public class ZombieEntity : MonoBehaviour { [Header("Configurações do Zumbi")] public NavMeshAgent agent; public Transform playerTarget; public float health = 100f; public float attackRange = 1.8f; public float attackDamage = 15f; public float attackRate = 1.5f; private float nextAttackTime = 0f; private bool isDead = false; void Start() { if (agent == null) agent = GetComponent(); if (playerTarget == null) { GameObject playerObj = GameObject.FindGameObjectWithTag("Player"); if (playerObj != null) playerTarget = playerObj.transform; } } void Update() { if (isDead || playerTarget == null) return; float distance = Vector3.Distance(transform.position, playerTarget.position); if (distance <= attackRange) { agent.isStopped = true; if (Time.time >= nextAttackTime) { ExecuteAttack(); nextAttackTime = Time.time + attackRate; } } else { agent.isStopped = false; agent.SetDestination(playerTarget.position); } } private void ExecuteAttack() { GameControllerMobile playerScript = playerTarget.GetComponent(); if (playerScript != null) { playerScript.PlayerTakeDamage(attackDamage); } } public void TakeDamage(float amount) { if (isDead) return; health -= amount; if (health <= 0f) { isDead = true; agent.enabled = false; GetComponent().enabled = false; Destroy(gameObject, 3.0f); } } } // ========================================== // CLASSE DE PORTAS E JANELAS (Anexe nos Objetos Interativos) // ========================================== public class InteractiveObject : MonoBehaviour { public float openAngle = 90f; public float speed = 3f; private bool isOpen = false; private Quaternion defaultRotation; private Quaternion openRotation; void Start() { defaultRotation = transform.localRotation; openRotation = Quaternion.Euler(transform.localEulerAngles + new Vector3(0, openAngle, 0)); } void Update() { Quaternion targetRotation = isOpen ? openRotation : defaultRotation; transform.localRotation = Quaternion.Slerp(transform.localRotation, targetRotation, Time.deltaTime * speed); } public void ToggleState() { isOpen = !isOpen; } }
