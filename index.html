function updateZombies(){ for(const z of zombies){ const dx = player.x - z.mesh.position.x; const dz = player.z - z.mesh.position.z; const dist = Math.hypot(dx,dz); /* ========================================= ZUMBI SEMPRE EM PÉ ========================================= */ z.mesh.position.y = 0; /* Faz o zumbi olhar somente para os lados. NÃO inclina para cima/baixo. */ const angle = Math.atan2(dx,dz); z.mesh.rotation.x = 0; z.mesh.rotation.y = angle; z.mesh.rotation.z = 0; /* ========================================= CAMINHANDO ========================================= */ if(dist > 2.7){ z.mesh.position.x += dx/dist * z.speed; z.mesh.position.z += dz/dist * z.speed; z.phase += .16; const walk = Math.sin(z.phase) * .7; /* pernas */ z.leftLeg.rotation.x = walk; z.rightLeg.rotation.x = -walk; /* braços */ z.leftArm.rotation.x = -walk * .55; z.rightArm.rotation.x = walk * .55; /* pequena movimentação, mas sem derrubar o zumbi */ z.mesh.rotation.z = 0; /* balanço leve da cabeça/corpo */ z.mesh.position.y = 0; } /* ========================================= ATAQUE ========================================= */ else{ /* GARANTE QUE ELE NÃO DEITE */ z.mesh.position.y = 0; z.mesh.rotation.x = 0; z.mesh.rotation.z = 0; /* mantém olhando para o jogador */ z.mesh.rotation.y = angle; z.attackTimer++; const attackMove = Math.sin(z.attackTimer * .35) * .18; /* braços levantados */ z.leftArm.rotation.x = -1.0 + attackMove; z.rightArm.rotation.x = -1.0 - attackMove; /* pernas ficam normais */ z.leftLeg.rotation.x = 0; z.rightLeg.rotation.x = 0; /* corpo não inclina */ z.mesh.rotation.z = 0; /* ataque */ if(z.attackTimer % 70 === 0){ zombieSound(); player.health -= 7; damageEffect(); if(player.health <= 0){ gameOver(); } } } } }
